const GAS_API_URL = "https://script.google.com/macros/s/AKfycb.../exec";


// 1. 초기 설정 로드 (initialize 함수 내부)

function initialize() {

  bindEvents();

  // fetch를 통해 구글 스프레드시트 설정 가져오기

    .then(res => res.json())

// 2. 출결 제출 (submitAttendance 함수 내부)

async function submitAttendance(action) {

  if (state.submitting || !state.config || !validateForm()) return;


  state.submitting = true;

  setSubmittingState(true, action);


  const payload = {

    className: state.selectedClass,

    name: elements.nameInput.value.trim(),

    phoneLast4: elements.phoneInput.value.replace(/\D/g, ''),

    dailyCode: state.config.requireDailyCode ? elements.codeInput.value.trim() : '',

    action: action,

  };


  try { 

    payload.position = await spLocate(); 

  } catch (error) { 

    state.submitting = false; 

    setSubmittingState(false); 

    showToast(getErrorMessage(error)); 

    return; 

  }


  // 구글 Apps Script로 POST 요청 전송

  try {

    const response = await fetch(GAS_API_URL, {

      method: "POST",

      // GAS 리다이렉트 처리 및 CORS 우회를 위해 text/plain 전송 권장

      headers: { "Content-Type": "text/plain;charset=utf-8" },

      body: JSON.stringify(payload),

    });

    const result = await response.json();


    state.submitting = false;

    setSubmittingState(false);


    if (result.ok) {

      handleSuccess(result.data);

    } else {

      showToast(result.error || "오류가 발생했습니다.");

    }

  } catch (err) {

    state.submitting = false;

    setSubmittingState(false);

    showToast("서버 통신 오류가 발생했습니다. 다시 시도해 주세요.");

  }

}

Wi-Fi와 위치 서비스를 켜고 다시 시도해 주세요.':'현재 위치를 확인할 수 없습니다. 다시 시도해 주세요.')),{enableHighAccuracy:true,maximumAge:0,timeout:15000});request();});}
